-- ===== 1) BUCKET PÚBLICO ===== insert into storage.buckets (id, name, public) values ('memorial', 'memorial', true) on conflict (id) do update set public = true; -- ===== 2) TABELA ===== create table if not exists public.memorial_data ( id uuid primary key default gen_random_uuid(), tipo text, titulo text, descricao text, data text, legenda text, nome text, relacao text, trecho text, link text, arquivo_url text, criado_em timestamptz default now() ); -- ===== 3) PERMISSÕES DA TABELA (visitante anônimo lê e escreve) ===== alter table public.memorial_data enable row level security; drop policy if exists "md_select" on public.memorial_data; create policy "md_select" on public.memorial_data for select to anon, authenticated using (true); drop policy if exists "md_insert" on public.memorial_data; create policy "md_insert" on public.memorial_data for insert to anon, authenticated with check (true); drop policy if exists "md_update" on public.memorial_data; create policy "md_update" on public.memorial_data for update to anon, authenticated using (true) with check (true); drop policy if exists "md_delete" on public.memorial_data; create policy "md_delete" on public.memorial_data for delete to anon, authenticated using (true); -- ===== 4) PERMISSÕES DO STORAGE ===== drop policy if exists "mem_sel" on storage.objects; create policy "mem_sel" on storage.objects for select to anon, authenticated using (bucket_id = 'memorial'); drop policy if exists "mem_ins" on storage.objects; create policy "mem_ins" on storage.objects for insert to anon, authenticated with check (bucket_id = 'memorial'); drop policy if exists "mem_upd" on storage.objects; create policy "mem_upd" on storage.objects for update to anon, authenticated using (bucket_id = 'memorial') with check (bucket_id = 'memorial'); drop policy if exists "mem_del" on storage.objects; create policy "mem_del" on storage.objects for delete to anon, authenticated using (bucket_id = 'memorial');
